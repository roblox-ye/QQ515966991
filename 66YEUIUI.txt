local function setting(key)
    local value, fallback = UI_CONFIG[key], UI_DEFAULTS[key]
    if fallback ~= nil and type(value) ~= type(fallback) then return fallback end
    if type(value) == "number" and (value ~= value or math.abs(value) == math.huge) then return fallback end
    return value
end
if type(UI_CONFIG.Scale) ~= "number" or UI_CONFIG.Scale ~= UI_CONFIG.Scale or UI_CONFIG.Scale <= 0 or math.abs(UI_CONFIG.Scale) == math.huge then
    UI_CONFIG.Scale = DefaultScale
end
local function SaveConfig()
    if not writefile then return false, "执行器不支持配置保存" end
    local success, result = pcall(function() writefile(ConfigPath, HttpService:JSONEncode(UI_CONFIG)) end)
    if success then savedConfig = table.clone(UI_CONFIG) end
    return success, result
end

    if not writefile then return false end
    local data = table.clone(savedConfig)
    for _, key in ipairs({"WindowWidth", "WindowHeight", "WindowOffsetX", "WindowOffsetY"}) do data[key] = UI_CONFIG[key] end
    local success = pcall(function() writefile(ConfigPath, HttpService:JSONEncode(data)) end)
    if success then savedConfig = data end
    return success
end

-- Native tweens do the interpolation; this owner only coordinates replacement and cleanup.
local UIManagers = {}
local latestUI
local function finite(value)
    return type(value) == "number" and value == value and math.abs(value) < math.huge
end
local function assign(obj, key, value)
    if obj[key] ~= value then obj[key] = value; return true end
    return false
end
local function managerFor(obj)
    while obj do
        if UIManagers[obj] then return UIManagers[obj] end
        obj = obj.Parent
    end
end
local Motion = {}
Motion.__index = Motion
function Motion:Connect(owner, signal, callback)
    local connection
    connection = signal:Connect(function(...)
        if self.Alive and (owner == self.Root or owner.Parent) then callback(...) end
    end)
    self.Connections[connection] = owner
    return connection
end
function Motion:Disconnect(connection)
    if connection then self.Connections[connection] = nil; connection:Disconnect() end
end
function Motion:Spawn(owner, callback, delay)
    if not self.Alive then return end
    local entry = {Owner = owner}
    self.Tasks[entry] = true
    local function run()
        local success, err = pcall(function()
            if self.Alive and (owner == self.Root or owner.Parent) then callback() end
        end)
        self.Tasks[entry] = nil
        if not success then error(err, 0) end
    end
    entry.Thread = delay and task.delay(delay, run) or task.defer(run)
    return entry
end
function Motion:RetireTween(tween)
    self.RetiredTweens[tween] = true
    self:Spawn(self.Root, function()
        self.RetiredTweens[tween] = nil
        tween:Destroy()
    end)
end
function Motion:CreateWindow(obj, info, goals)
    -- Keep handles referenced by the frozen cancellation loop valid until that loop releases them.
    for previous, kind in pairs(self.RetiredTweens) do
        if kind == "window" then
            local referenced = false
            for _, active in ipairs(_G.activeUITweens or {}) do if active == previous then referenced = true end end
            if not referenced then self.RetiredTweens[previous] = nil; previous:Destroy() end
        end
    end
    -- Exactly the original native Create call: no cancellation, duration or goal adjustment.
    local tween = TweenService:Create(obj, info, goals)
    if obj == self.MainScale then self.WindowScaleTween = tween end
    local connection
    connection = tween.Completed:Connect(function()
        self.WindowTweens[tween] = nil
        connection:Disconnect()
        self.RetiredTweens[tween] = "window"
        if self.WindowScaleTween == tween then
            self.WindowScaleTween = nil
            if self.Alive and self.ClampLayout then self.ClampLayout() end
        end
    end)
    self.WindowTweens[tween] = connection
    return tween
end
function Motion:ClearKeyCapture()
    local capture = self.KeyCapture
    self.KeyCapture = nil
    if capture and capture.Cancel then capture.Cancel() end
end
function Motion:SetKeyCapture(owner, cancel)
    self:ClearKeyCapture()
    self.KeyCapture = {Owner = owner, Cancel = cancel}
end
function Motion:BindKey(owner, handler)
    self.Keys[owner] = handler
    if self.KeyConnection then return end
    self.KeyConnection = self:Connect(self.Root, UserInputService.InputBegan, function(input, processed)
        if processed or input.UserInputType ~= Enum.UserInputType.Keyboard then return end
        local capture = self.KeyCapture
        if capture then
            local current = self.Keys[capture.Owner]
            if current then current(input, true) end
            return
        end
        for object, current in pairs(self.Keys) do
            local callback, argument = current(input, false)
            if callback then task.spawn(callback, argument) end
        end
    end)
end
function Motion:Cancel(obj, finish)
    local entry = self.Animations[obj]
    if not entry then return end
    self.Animations[obj] = nil
    entry.Connection:Disconnect()
    entry.Tween:Cancel()
    if finish and obj.Parent then
        for key, value in pairs(entry.Goals) do assign(obj, key, value) end
    end
    entry.Tween:Destroy()
end
function Motion:IsVisible(obj)
    if not self.Alive or not self.Root.Enabled then return false end
    while obj and obj ~= self.Root do
        if obj:IsA("GuiObject") and not obj.Visible then return false end
        obj = obj.Parent
    end
    return obj == self.Root
end
function Motion:Create(obj, info, goals)
    self:Cancel(obj, false)
    -- Hidden controls receive their final state without a running visual animation.
    if not self:IsVisible(obj) then info = TweenInfo.new(0) end
    local tween = TweenService:Create(obj, info, goals)
    local entry = {Tween = tween, Goals = goals}
    self.Animations[obj] = entry
    entry.Connection = tween.Completed:Connect(function()
        if self.Animations[obj] == entry then self.Animations[obj] = nil end
        entry.Connection:Disconnect()
        -- Consumers may subscribe to Completed, so destroy after signal dispatch.
        self:RetireTween(tween)
    end)
    return tween
end
function Motion:StopVisuals(root)
    local objects = {}
    for obj in pairs(self.Animations) do
        if obj == root or obj:IsDescendantOf(root) then table.insert(objects, obj) end
    end
    for _, obj in ipairs(objects) do self:Cancel(obj, true) end
end
function Motion:EndPointer(released)
    local pointer = self.Pointer
    if not pointer then return end
    self.Pointer = nil
    self:Disconnect(pointer.Move)
    self:Disconnect(pointer.End)
    if pointer.Finish then pointer.Finish(released) end
end
function Motion:BeginPointer(owner, input, move, finish)
    self:EndPointer(false)
    local pointer = {Owner = owner, Input = input, Finish = finish}
    self.Pointer = pointer
    pointer.Move = self:Connect(owner, UserInputService.InputChanged, function(changed)
        if input.UserInputType == Enum.UserInputType.Touch then
            if changed == input then move(changed) end
        elseif changed.UserInputType == Enum.UserInputType.MouseMovement then
            move(changed)
        end
    end)
    pointer.End = self:Connect(owner, UserInputService.InputEnded, function(ended)
        if ended == input or (input.UserInputType == Enum.UserInputType.MouseButton1 and ended.UserInputType == Enum.UserInputType.MouseButton1) then
            self:EndPointer(true)
        end
    end)
end
function Motion:Remove(obj)
    self:Cancel(obj, false)
    self.ThemeObjects[obj] = nil
    self.PendingTheme[obj] = nil
    self.Feedback[obj] = nil
    local removedControl = self.Controls[obj] or self.Sections[obj]
    self.Controls[obj] = nil
    self.Sections[obj] = nil
    self.Keys[obj] = nil
    if self.KeyCapture and self.KeyCapture.Owner == obj then self:ClearKeyCapture() end
    if self.KeyConnection and not next(self.Keys) then self:Disconnect(self.KeyConnection); self.KeyConnection = nil end
    if removedControl then self.SearchDirty = true; self:QueueSearch() end
    if obj == self.Stats or obj == self.Main then
        if self.FrameConnection then self.FrameConnection:Disconnect(); self.FrameConnection = nil end
    end
    for i = #self.Tabs, 1, -1 do
        local tab = self.Tabs[i]
        if tab[1] == obj or tab[2] == obj then
            table.remove(self.Tabs, i)
            if self.BeforeSearch == tab then self.BeforeSearch = nil end
            if library.currentTab == tab then library.currentTab = nil end
        end
    end
    for connection, owner in pairs(self.Connections) do
        if owner == obj then self:Disconnect(connection) end
    end
    for entry in pairs(self.Tasks) do
        if entry.Owner == obj then
            self.Tasks[entry] = nil
            if entry.Thread and coroutine.status(entry.Thread) ~= "dead" and entry.Thread ~= coroutine.running() then pcall(task.cancel, entry.Thread) end
        end
    end
    if self.Pointer and self.Pointer.Owner == obj then self:EndPointer(false) end
    for i = #self.Notifications, 1, -1 do
        if self.Notifications[i] == obj then table.remove(self.Notifications, i) end
    end
end
function Motion:Destroy()
    if not self.Alive then return end
    self.Alive = false
    self:ClearKeyCapture()
    self:EndPointer(false)
    if self.FrameConnection then self.FrameConnection:Disconnect(); self.FrameConnection = nil end
    for connection in pairs(self.Connections) do connection:Disconnect() end
    table.clear(self.Connections)
    for obj in pairs(self.Animations) do self:Cancel(obj, false) end
    for tween, connection in pairs(self.WindowTweens) do
        connection:Disconnect()
        tween:Cancel()
        tween:Destroy()
    end
    table.clear(self.WindowTweens)
    for tween in pairs(self.RetiredTweens) do tween:Destroy() end
    table.clear(self.RetiredTweens)
    for entry in pairs(self.Tasks) do
        if entry.Thread and coroutine.status(entry.Thread) ~= "dead" and entry.Thread ~= coroutine.running() then pcall(task.cancel, entry.Thread) end
    end
    table.clear(self.Tasks)
    table.clear(self.ThemeObjects)
    table.clear(self.PendingTheme)
    table.clear(self.Feedback)
    table.clear(self.Controls)
    table.clear(self.Keys)
    table.clear(self.Sections)
    table.clear(self.Tabs)
    table.clear(self.Notifications)
    UIManagers[self.Root] = nil
    if latestUI == self then latestUI = nil end
    library.currentTab = nil
    for _, tween in ipairs(_G.activeUITweens or {}) do
        pcall(function() tween:Cancel(); tween:Destroy() end)
    end
    if _G.activeUITweens then table.clear(_G.activeUITweens) end
end
function Motion.new(root)
    local self = setmetatable({Root = root, Alive = true, Connections = {}, Animations = {}, Tasks = {}, WindowTweens = {}, RetiredTweens = {},
        ThemeObjects = {}, PendingTheme = {}, Feedback = {}, Controls = {}, Keys = {}, Sections = {}, Tabs = {}, Notifications = {},
        ThemePasses = 0, ThemeWrites = 0, StatisticsUpdates = 0}, Motion)
    UIManagers[root] = self
    latestUI = self
    self:Connect(root, root.DescendantAdded, function(obj) self:QueueTheme(obj) end)
    self:Connect(root, root.DescendantRemoving, function(obj) self:Remove(obj) end)
    self:Connect(root, root.Destroying, function() self:Destroy() end)
    self:Connect(root, root:GetPropertyChangedSignal("Enabled"), function() self:SyncVisibility() end)
    return self
end
function Motion:Write(obj, key, value)
    if assign(obj, key, value) then self.ThemeWrites = self.ThemeWrites + 1 end
end
function Motion:GetTheme()
    local color = FontColorPresets[setting("FontColorMode")] or FontColorPresets["纯净洁白"]
    local dark = color.R * 0.2126 + color.G * 0.7152 + color.B * 0.0722 > 0.55
    local palette = Palettes[setting("Palette")] or Palettes["赛博朋克 (Cyberpunk)"]
    return {Text = color, Font = FontTypes[setting("FontMode")] or Enum.Font.GothamBold,
        Size = finite(setting("FontSizeOffset")) and setting("FontSizeOffset") or 0,
        Surface = dark and Color3.fromRGB(25, 30, 41) or Color3.fromRGB(248, 249, 252),
        Raised = dark and Color3.fromRGB(38, 45, 59) or Color3.fromRGB(230, 234, 241),
        Base = dark and Color3.fromRGB(17, 21, 30) or Color3.fromRGB(240, 243, 249),
        Accent = palette.Keypoints[1].Value, Palette = palette, Dark = dark}
end
local SurfaceNames = {BasePlate = true, ReadabilityVeil = true, Side = true, SB = true,
    ConfirmMain = true, Open = true, NotificationCard = true, BoxBG = true, SliderValBG = true,
    SliderTrack = true, SliderFill = true, ToggleDisable = true, TopLine = true, Indicator = true, TabBtn = true}
local function themeRelevant(obj)
    return obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") or obj:IsA("UIGradient")
        or obj:IsA("UIStroke") or (obj:IsA("GuiObject") and (obj:GetAttribute("IsControlBG") or SurfaceNames[obj.Name]))
end
function Motion:QueueTheme(obj)
    self.PendingTheme[obj] = true
    self:RefreshTheme()
end
function Motion:ApplyTheme(obj, theme, font, color, size, surface, palette)
    if not obj.Parent then return end
    if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
        local base = obj:GetAttribute("BaseTextSize")
        if not base then base = obj.TextSize; obj:SetAttribute("BaseTextSize", base) end
        if obj.Name ~= "Icon" and obj.Name ~= "Crown" then
            if font and obj.Name ~= "StatsText" then self:Write(obj, "Font", theme.Font) end
            if color then
                self:Write(obj, "TextColor3", theme.Text)
                self:Write(obj, "TextStrokeColor3", theme.Dark and Color3.new(0, 0, 0) or Color3.new(1, 1, 1))
                self:Write(obj, "TextStrokeTransparency", self.HasBackground and 0.75 or 1)
                if obj:IsA("TextBox") then self:Write(obj, "PlaceholderColor3", theme.Text:Lerp(theme.Surface, 0.38)) end
            end
            if size then self:Write(obj, "TextSize", math.clamp(base + theme.Size, 9, 100)) end
        end
    end
    if surface then
        if obj:GetAttribute("IsControlBG") then
            local feedback = self.Feedback[obj]
            local amount = feedback and (feedback.Pressed and 0.12 or feedback.Hovered and 0.06 or 0) or 0
            if obj:GetAttribute("UIState") then amount = math.max(amount, 0.09) end
            self:Cancel(obj, false)
            self:Write(obj, "BackgroundColor3", theme.Surface:Lerp(theme.Accent, amount))
            self:Write(obj, "BackgroundTransparency", interfaceTransparency)
        elseif obj.Name == "BasePlate" or obj.Name == "ReadabilityVeil" then
            self:Write(obj, "BackgroundColor3", theme.Base)
        elseif obj.Name == "Side" or obj.Name == "SB" then
            self:Write(obj, "BackgroundColor3", theme.Surface)
        elseif obj.Name == "ConfirmMain" or obj.Name == "Open" or obj.Name == "NotificationCard" then
            self:Write(obj, "BackgroundColor3", theme.Surface)
        elseif obj.Name == "BoxBG" or obj.Name == "SliderValBG" or obj.Name == "SliderTrack" or obj.Name == "GlobalSearch" then
            self:Write(obj, "BackgroundColor3", theme.Raised)
        elseif obj.Name == "ToggleDisable" then
            self:Cancel(obj, false)
            self:Write(obj, "BackgroundColor3", obj:GetAttribute("UIState") and theme.Accent or theme.Raised)
        elseif obj.Name == "TopLine" then
            self:Write(obj, "BackgroundColor3", theme.Text)
        elseif obj:IsA("UIStroke") and obj.Parent:GetAttribute("IsControlBG") then
            self:Write(obj, "Color", theme.Text)
            self:Write(obj, "Transparency", 0.84)
        elseif obj:IsA("UIGradient") and (obj.Parent.Name == "Side" or obj.Parent.Name == "SB") then
            self:Write(obj, "Color", ColorSequence.new(Color3.new(1, 1, 1)))
        end
    end
    if palette then
        if obj.Name == "Indicator" or obj.Name == "TabBtn" or obj.Name == "SliderFill" then self:Write(obj, "BackgroundColor3", theme.Accent) end
        if obj:IsA("UIGradient") and obj.Parent:IsA("UIStroke") then
            self:Write(obj, "Color", theme.Palette)
        elseif obj:IsA("UIGradient") and obj.Parent.Name == "ScriptTitle" then
            self:Write(obj, "Color", ColorSequence.new(Color3.new(1, 1, 1)))
        end
    end
end
function Motion:RefreshTheme()
    if self.ThemeQueued or not self.Alive then return end
    self.ThemeQueued = true
    self:Spawn(self.Root, function()
        self.ThemeQueued = false
        local theme = self:GetTheme()
        local previous = self.LastTheme
        local font = not previous or previous.Font ~= theme.Font
        local size = not previous or previous.Size ~= theme.Size
        local color = not previous or previous.Text ~= theme.Text or previous.Background ~= self.HasBackground
        local palette = not previous or previous.Palette ~= theme.Palette
        local surface = color or palette or not previous or previous.Alpha ~= interfaceTransparency
        if font or size or color or surface or palette then
            self.ThemePasses = self.ThemePasses + 1
            for obj in pairs(self.ThemeObjects) do self:ApplyTheme(obj, theme, font, color, size, surface, palette) end
        end
        for obj in pairs(self.PendingTheme) do
            if obj.Parent and themeRelevant(obj) then
                self.ThemeObjects[obj] = true
                self:ApplyTheme(obj, theme, true, true, true, true, true)
            end
        end
        table.clear(self.PendingTheme)
        theme.Background = self.HasBackground
        theme.Alpha = interfaceTransparency
        self.LastTheme = theme
        if self.Shadow then
            local intensity = finite(setting("GlowIntensity")) and math.clamp(setting("GlowIntensity"), 0, 100) or 50

function Motion:FeedbackState(obj, state)
    if not obj.Parent then return end
    local theme = self:GetTheme()
    local amount = state.Pressed and 0.12 or state.Hovered and 0.06 or 0
    if obj:GetAttribute("UIState") then amount = math.max(amount, 0.09) end
    if obj:GetAttribute("IsControlBG") then
        self:Create(obj, TweenInfo.new(state.Pressed and 0.08 or 0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = theme.Surface:Lerp(theme.Accent, amount)}):Play()
    end
    local stroke = obj:FindFirstChildOfClass("UIStroke")
    if stroke then self:Create(stroke, TweenInfo.new(0.14, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Transparency = state.Pressed and 0.3 or state.Hovered and 0.52 or 0.84}):Play() end
end
function Motion:BindFeedback(obj)
    if self.Feedback[obj] then return end
    local state = {Hovered = false, Pressed = false}
    self.Feedback[obj] = state
    self:Connect(obj, obj.MouseEnter, function() state.Hovered = true; self:FeedbackState(obj, state) end)
    self:Connect(obj, obj.MouseLeave, function() state.Hovered = false; state.Pressed = false; self:FeedbackState(obj, state) end)
    self:Connect(obj, obj.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then state.Pressed = true; self:FeedbackState(obj, state) end
    end)
    self:Connect(obj, obj.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then state.Pressed = false; self:FeedbackState(obj, state) end
    end)
end
function Motion:SyncVisibility()
    if not self.Alive or not self.Main or not self.Stats then return end
    local visible = self:IsVisible(self.Stats)
    if not self.Root.Enabled or not self.Main.Visible then
        self:ClearKeyCapture()
        self:EndPointer(false)
        self:StopVisuals(self.Main)
    end
    if not visible and self.FrameConnection then self.FrameConnection:Disconnect(); self.FrameConnection = nil end
    if not visible then
        self.Frames = 0; self.Elapsed = 0
        return
    end
    if not setting("MarqueeEnabled") then assign(self.Gradient, "Rotation", 45) end
    if self.FrameConnection then return end
    self.Frames = 0; self.Elapsed = 0
    self.FrameConnection = RunService.RenderStepped:Connect(function(dt)
        self.Frames = self.Frames + 1
        self.Elapsed = self.Elapsed + dt
        if setting("MarqueeEnabled") then
            self.Gradient.Rotation = (os.clock() * (finite(setting("MarqueeSpeed")) and setting("MarqueeSpeed") or 1) * 60) % 360
        end
        if self.Elapsed < 1 then return end
        local fps = math.floor(self.Frames / self.Elapsed)
        self.Frames = 0; self.Elapsed = 0
        local ping = 0
        pcall(function() ping = math.floor(game:GetService("Stats").Network.ServerStatsItem["Data Ping"]:GetValue()) end)
        local text = string.format("FPS: %d\nPing: %dms\n时间: %s", fps, ping, os.date("%H:%M:%S"))
        if assign(self.Stats, "Text", text) then self.StatisticsUpdates = self.StatisticsUpdates + 1 end
    end)
end

function Motion:QueueSearch()
    if self.SearchQueued or not self.Search then return end
    self.SearchQueued = true
    self:Spawn(self.Root, function()
        self.SearchQueued = false
        local query = string.lower(self.Search.Text):match("^%s*(.-)%s*$")
        if query == (self.SearchQuery or "") and not self.SearchDirty then return end
        self.SearchDirty = false
        if query ~= "" and (self.SearchQuery or "") == "" then self.BeforeSearch = library.currentTab end
        self.SearchQuery = query
        local hits = {}
        for module, entry in pairs(self.Controls) do
            local title = module:FindFirstChild("ControlTitle", true) or module:FindFirstChildOfClass("TextLabel") or module:FindFirstChildOfClass("TextButton")
            local displayed = title and string.lower(title.Text) or ""
            local match = query == "" or string.find(displayed, query, 1, true) ~= nil or string.find(entry.Text, query, 1, true) ~= nil or string.find(entry.Section.Title, query, 1, true) ~= nil
            module.Visible = match
            if match then hits[entry.Section] = true end
        end
        local tabs = {}
        for _, section in pairs(self.Sections) do
            local match = query == "" or hits[section] == true
            section.Frame.Visible = match
            if query == "" then section.Objects.Visible = section.Open else section.Objects.Visible = match end
            if match then tabs[section.Tab] = true end
        end
        local first
        for _, tab in ipairs(self.Tabs) do
            tab[1].Visible = query == "" or tabs[tab] == true
            if tab[1].Visible and not first then first = tab end
        end
        if query == "" and self.BeforeSearch and self.BeforeSearch[1].Parent then
            switchTab(self.BeforeSearch); self.BeforeSearch = nil
        elseif first and (not library.currentTab or not library.currentTab[1].Visible) then switchTab(first) end
    end)
end

function switchTab(new)
    if not new or not new[2].Parent then return end
    local old = library.currentTab
    if old == new then return end
    local owner = managerFor(new[2])
    if not owner then return end
    local function select(tab, selected)
        local indicator = tab[1]:FindFirstChild("Indicator")
        local button = tab[1]:FindFirstChild("TabBtn")
        local title = tab[1]:FindFirstChild("TabText")
        if indicator then owner:Create(indicator, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = selected and 0 or 1}):Play() end
        if button then owner:Create(button, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = selected and 0.87 or 1}):Play() end
        if title then owner:Create(title, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {TextTransparency = selected and 0 or 0.22}):Play() end
    end
    library.currentTab = new
    if old and old[2].Parent then
        owner:StopVisuals(old[2])
        old[2].Visible = false
        select(old, false)
    end
    new[2].Visible = true
    new[2].Position = UDim2.new(0, 0, 0, 6)
    select(new, true)
    owner:Create(new[2], TweenInfo.new(0.18, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {Position = UDim2.new()}):Play()
end

function drag(frame, hold, onReleased)
    hold = hold or frame
    local owner = managerFor(frame)
    if not owner then return end
    hold.Active = true
    owner:Connect(hold, hold.InputBegan, function(input)
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        local start, position = input.Position, frame.Position
        local bounds = owner.Main and owner.Main.AbsolutePosition
        local viewport = frame.Name == "WindowLayout" and frame.AbsoluteSize
        local gripX, gripY = bounds and hold.AbsolutePosition.X - bounds.X, bounds and hold.AbsolutePosition.Y - bounds.Y
        local gripWidth = hold.AbsoluteSize.X
        local visibleGrip = math.min(24, gripWidth)
        owner:BeginPointer(hold, input, function(changed)
            local delta = changed.Position - start
            local x, y = position.X.Offset + delta.X, position.Y.Offset + delta.Y
            if viewport and viewport.X > 0 and viewport.Y > 0 then
                local minX = visibleGrip - gripX - gripWidth
                local maxX = viewport.X - visibleGrip - gripX
                local minY = -gripY
                local maxY = math.max(minY, viewport.Y - 24 - gripY)
                x = position.X.Offset + math.clamp(bounds.X + delta.X, math.min(minX, maxX), maxX) - bounds.X
                y = position.Y.Offset + math.clamp(bounds.Y + delta.Y, minY, maxY) - bounds.Y
            end
            assign(frame, "Position", UDim2.new(position.X.Scale, x, position.Y.Scale, y))
        end, function(released) if released and onReleased then onReleased() end end)
    end)
end

    local function ClampLayout()
        local viewport = WindowLayout.AbsoluteSize
        if viewport.X <= 0 or viewport.Y <= 0 or motion.WindowScaleTween then return end
        local scale = uiScale.Scale > 0 and uiScale.Scale or UI_CONFIG.Scale
        local width, height = CloudMain.Size.X.Offset, CloudMain.Size.Y.Offset
        local chain, cursor = {}, ScriptTitle
        while cursor and cursor ~= CloudMain do
            table.insert(chain, 1, cursor)
            cursor = cursor.Parent
        end
        if cursor ~= CloudMain then return end
        local gripLeft, gripTop, gripWidth, gripHeight = 0, 0, width, height
        for _, object in ipairs(chain) do
            local nextWidth = gripWidth * object.Size.X.Scale + object.Size.X.Offset
            local nextHeight = gripHeight * object.Size.Y.Scale + object.Size.Y.Offset
            gripLeft = gripLeft + gripWidth * object.Position.X.Scale + object.Position.X.Offset - nextWidth * object.AnchorPoint.X
            gripTop = gripTop + gripHeight * object.Position.Y.Scale + object.Position.Y.Offset - nextHeight * object.AnchorPoint.Y
            gripWidth, gripHeight = nextWidth, nextHeight
        end
        gripLeft, gripTop, gripWidth = gripLeft * scale, gripTop * scale, gripWidth * scale
        local visibleGrip = math.min(24, gripWidth)
        local baseX, baseY = viewport.X * 0.5 - width * scale * 0.5, viewport.Y * 0.4 - height * scale * 0.5
        local minX = visibleGrip - gripLeft - gripWidth - baseX
        local maxX = viewport.X - visibleGrip - gripLeft - baseX
        local minY = -gripTop - baseY
        local maxY = math.max(minY, viewport.Y - 24 - gripTop - baseY)
        local position = WindowLayout.Position
        assign(WindowLayout, "Position", UDim2.new(0, math.clamp(position.X.Offset, math.min(minX, maxX), maxX), 0, math.clamp(position.Y.Offset, minY, maxY)))
    end
    motion.ClampLayout = ClampLayout
    motion:Connect(WindowLayout, WindowLayout:GetPropertyChangedSignal("AbsoluteSize"), ClampLayout)
    motion:Connect(CloudMain, CloudMain:GetPropertyChangedSignal("Size"), ClampLayout)
    motion:Connect(uiScale, uiScale:GetPropertyChangedSignal("Scale"), ClampLayout)
    ClampLayout()
    motion:Spawn(WindowLayout, ClampLayout)
    drag(WindowLayout, ScriptTitle, PersistLayout)

                SliderHit.Active = true
                SliderHit.ZIndex = SliderBack.ZIndex + 2
                SliderHit.Parent = SliderBack
                motion:Connect(SliderHit, SliderHit.InputBegan, function(input)
                    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
                    local scrolled = Tab.ScrollingEnabled
                    Tab.ScrollingEnabled = false
                    funcs:SetValue(nil, input.Position.X)
                    motion:BeginPointer(SliderHit, input, function(changed) funcs:SetValue(nil, changed.Position.X) end, function(released)
                        if Tab.Parent then Tab.ScrollingEnabled = scrolled end
                        if released and onReleased then onReleased(currentValue) end
                    end)
                end)
                motion:Connect(SliderValue, SliderValue.Focused, function() boxFocused = true end)
                motion:Connect(SliderValue, SliderValue.FocusLost, function()
                    boxFocused = false
                    if SliderValue.Text == "" then funcs:SetValue(default) else writeValue(currentValue) end
                    if onReleased then onReleased(currentValue) end
                end)
                motion:Connect(SliderValue, SliderValue:GetPropertyChangedSignal("Text"), function()
                    if not boxFocused or updatingText then return end
                    local value = tonumber(SliderValue.Text)
                    if finite(value) then funcs:SetValue(value, nil, true) end
                end)

                RefreshGlobalUI()
                return funcs
            end

                local OptionList = Instance.new("ScrollingFrame")
                OptionList.Name = "OptionList"
                OptionList.BackgroundTransparency = 1
                OptionList.BorderSizePixel = 0
                OptionList.Size = UDim2.new(1, 0, 0, 0)
                OptionList.AutomaticCanvasSize = Enum.AutomaticSize.Y
                OptionList.CanvasSize = UDim2.new()
                OptionList.ScrollBarThickness = 3
                OptionList.ClipsDescendants = true
                OptionList.Visible = false
                OptionList.Parent = DropdownModule
                local optionLayout = Instance.new("UIListLayout", OptionList)
                optionLayout.Padding = UDim.new(0, 6)
                optionLayout.SortOrder = Enum.SortOrder.LayoutOrder
                local open, focused = false, false
                local function ResizeOptions()
                    local count = 0
                    local query = focused and string.lower(DropdownText.Text) or ""
                    for _, option in ipairs(OptionList:GetChildren()) do
                        if option:IsA("TextButton") then
                            option.Visible = query == "" or string.find(string.lower(option.Text), query, 1, true) ~= nil
                            if option.Visible then count = count + 1 end
                        end
                    end
                    if open then OptionList.Visible = true end
                    local tween = motion:Create(OptionList, TweenInfo.new(0.18, Enum.EasingStyle.Quint, Enum.EasingDirection.Out), {Size = UDim2.new(1, 0, 0, open and math.min(192, count * 32) or 0)})
                    tween.Completed:Once(function(state) if state == Enum.PlaybackState.Completed and not open then OptionList.Visible = false end end)
                    tween:Play()
                end
                local function ToggleDropVis()
                    open = not open
                    DropdownOpen.Text = open and "−" or "+"
                    motion:Create(DropdownOpen, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Rotation = open and 90 or 0}):Play()
                    ResizeOptions()
                end
                motion:Connect(DropdownOpen, DropdownOpen.MouseButton1Click, ToggleDropVis)
                motion:Connect(DropdownTop, DropdownTop.MouseButton1Click, ToggleDropVis)
                DropdownText.ClearTextOnFocus = false
                motion:Connect(DropdownText, DropdownText.Focused, function() focused = true; if not open then ToggleDropVis() else ResizeOptions() end end)
                motion:Connect(DropdownText, DropdownText.FocusLost, function() focused = false; ResizeOptions() end)
                motion:Connect(DropdownText, DropdownText:GetPropertyChangedSignal("Text"), function() if focused then ResizeOptions() end end)

                local funcs = {}
                funcs.AddOption = function(self, option)
                    local Option = Instance.new("TextButton")
                    local OptionC = Instance.new("UICorner")

                    Option.Name = "Option_" .. option
                    Option.Parent = OptionList
                    Option.BackgroundColor3 = zyColor
                    Option.Size = UDim2.new(1, 0, 0, 26)
                    Option.AutoButtonColor = false
                    Option.Font = Enum.Font.GothamSemibold
                    Option.Text = option
                    Option.TextColor3 = TextColor
                    Option.TextSize = 12.000
                    Option.Visible = true
                    Option.BackgroundTransparency = interfaceTransparency
                    Option:SetAttribute("BaseTextSize", 12)
                    Option:SetAttribute("IsControlBG", true)

                    OptionC.CornerRadius = UDim.new(0, 6)
                    OptionC.Parent = Option

                    motion:Connect(Option, Option.MouseButton1Click, function()
                        ToggleDropVis()
                        callback(Option.Text)
                        DropdownText.Text = Option.Text
                        if flag ~= nil then library.flags[flag] = Option.Text end
                    end)
                    motion:BindFeedback(Option)
                    if open then ResizeOptions() end
                    RefreshGlobalUI()
                end

                funcs.SetOptions = function(self, options)
                    for _, v in next, OptionList:GetChildren() do
                        if v.Name:match("Option_") then v:Destroy() end
                    end
                    for _, v in next, options do funcs:AddOption(v) end
                    ResizeOptions()
                end

                funcs.Refresh = funcs.SetOptions
                funcs.Update = funcs.SetOptions
                funcs.SetSelection = function(self, value)
                    DropdownText.Text = tostring(value or "")
                    if flag ~= nil then library.flags[flag] = value end
                end
                funcs:SetOptions(options)
                if buildingSettings then
                    local keys = {UIFontModeDropdown = "FontMode", UIFontColorDropdown = "FontColorMode", UIColorDropdown = "Palette", UIGlowColorDropdown = "GlowColor"}
                    if keys[flag] then funcs:SetSelection(UI_CONFIG[keys[flag]]) end
                end
                
                RefreshGlobalUI()
                return funcs
            end
            return section
        end
        return tab
    end

            local height = math.clamp(size.Y.Offset + delta.Y / scale, 250, 700)
            assign(CloudMain, "Size", UDim2.new(0, width, 0, height))

            if ResizeHandle.Parent then motion:Create(ResizeHandle, TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {TextTransparency = 0.4}):Play() end

    function window:BuildUISettingsTab()
        if self.SettingsTabBuilt then return end 
        self.SettingsTabBuilt = true
        buildingSettings = true 

        local tab = self:Tab("界面设定", "rbxassetid://6031280882")
        self.SettingsTabData = tab._UIData
        
        local sec2 = tab:section("全局字体自定义", true)
        
        sec2:Slider("全局字体大小调整", "UIFontSizeOffset", setting("FontSizeOffset"), -4, 10, false, function(v)
            if buildingSettings then return end
            UI_CONFIG.FontSizeOffset = v
            RefreshGlobalUI()
        end)
        
        sec2:Dropdown("全局字体样式", "UIFontModeDropdown", FontTypeNames, function(v)
            if buildingSettings then return end
            UI_CONFIG.FontMode = v
            RefreshGlobalUI()
        end)
        
        sec2:Dropdown("全局字体颜色", "UIFontColorDropdown", FontColorNames, function(v)
            if buildingSettings then return end
            UI_CONFIG.FontColorMode = v
            RefreshGlobalUI()
        end)

        local sec = tab:section("自定义UI风格", true)

        local lastSavedScale = UI_CONFIG.Scale
        local scaleSliderFuncs        
        scaleSliderFuncs = sec:Slider("整体界面缩放 (%)", "UIScaleSlider", UI_CONFIG.Scale * 100, 50, 200, false, function(v)
            if buildingSettings then return end
            uiScale.Scale = v / 100
        end, function(v)
            if buildingSettings then return end
            local currentScale = v / 100
            if currentScale == lastSavedScale then return end
            
            ShowConfirm("确定要使用这个缩放大小吗？(需点击底部保存才可生效)", function()
                lastSavedScale = currentScale
                UI_CONFIG.Scale = lastSavedScale
            end, function()
                scaleSliderFuncs:SetValue(lastSavedScale * 100)
                uiScale.Scale = lastSavedScale
            end)
        end)
        
        window.SyncScale = function(value)
            lastSavedScale = value
            buildingSettings = true
            scaleSliderFuncs:SetValue(value * 100)
            buildingSettings = false
            uiScale.Scale = value
        end
        sec:Label("提示：UI卡住可按 左Alt + S 键重置大小")

        sec:Toggle("开启随机背景图功能", "UseRandomBGToggle", setting("UseRandomBG"), function(v)
            if buildingSettings then return end
            UI_CONFIG.UseRandomBG = v
            UpdateBackgroundDisplay()
        end)

        sec:Toggle("开启边框流光", "UIMarqueeToggle", setting("MarqueeEnabled"), function(v)
            if buildingSettings then return end
            UI_CONFIG.MarqueeEnabled = v
            motion:SyncVisibility()
        end)
        
        sec:Toggle("启用边缘辉光特效", "UIGlowToggle", setting("GlowEnabled"), function(v)
            if buildingSettings then return end
            UI_CONFIG.GlowEnabled = v
            RefreshGlobalUI()
        end)

        sec:Slider("边缘辉光强度", "UIGlowIntensity", setting("GlowIntensity"), 0, 100, false, function(v)
            if buildingSettings then return end
            UI_CONFIG.GlowIntensity = v
            RefreshGlobalUI()
        end)

        sec:Dropdown("更改UI主题与跑马灯颜色", "UIColorDropdown", PaletteNames, function(v)
            if buildingSettings then return end
            UI_CONFIG.Palette = v
            if Palettes[v] then
                MainStrokeGradient.Color = Palettes[v]
                RefreshGlobalUI()
            end
        end)
        
        sec:Dropdown("更改边缘辉光颜色", "UIGlowColorDropdown", FontColorNames, function(v)
            if buildingSettings then return end
            UI_CONFIG.GlowColor = v
            if setting("GlowEnabled") then
                RefreshGlobalUI()
            end
        end)
        
        sec:Slider("跑马灯速度", "UIMarqueeSpeed", setting("MarqueeSpeed") * 10, 1, 50, false, function(v)
            if buildingSettings then return end
            UI_CONFIG.MarqueeSpeed = v / 10
        end)

        sec:Textbox("自定义背景图 (需填Image ID)", "CustomBGImage", setting("BGImage"), function(val)
            if buildingSettings then return end
            if val == "" or val == "0" then
                CustomBG.Image = ""
                UI_CONFIG.BGImage = ""
            else
                local finalId = val
                if string.match(val, "^%d+$") then
                    finalId = "rbxassetid://" .. val
                end
                CustomBG.Image = finalId
                UI_CONFIG.BGImage = finalId
            end
            UpdateBackgroundDisplay()
        end)

        end)

                RefreshGlobalUI()
                local previousTab = self.SettingsTabData
                local wasCurrent = library.currentTab == previousTab
                if wasCurrent then library.currentTab = nil end
                if previousTab then previousTab[1]:Destroy(); previousTab[2]:Destroy() end
                self.SettingsTabBuilt = false
                self:BuildUISettingsTab()
                SaveConfig()

                pcall(function()
                    library:Notify("UI 设置", "已清除配置文件，界面已恢复默认", 4)
                end)
            end)
        end)
        buildingSettings = false
    end

    motion:Spawn(dogent, function()
